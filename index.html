        function sendMessage() {
            const inputField = document.getElementById('user-input');
            const chatBox = document.getElementById('chat-box');
            const userText = inputField.value.trim();
            if (!userText) return;
            // 사용자 메시지 추가
            const userBubble = document.createElement('div');
            userBubble.className = "flex items-start justify-end space-x-3";
            userBubble.innerHTML = `
                <div class="bg-blue-600 text-white rounded-2xl px-4 py-3 max-w-xl 
text-sm leading-relaxed shadow-sm">
                    ${escapeHtml(userText)}
                </div>
                <div class="bg-slate-600 text-white rounded-full h-8 w-8 flex 
items-center justify-center font-bold text-sm shrink-0">나</div>
            `;
            chatBox.appendChild(userBubble);
            inputField.value = '';
            chatBox.scrollTop = chatBox.scrollHeight;
            // 가상의 봇 응답 생성 (모의 시연)
            setTimeout(() => {
                const botBubble = document.createElement('div');
                botBubble.className = "flex items-start space-x-3";
                
                let responseContent = "질문하신 내용에 대한 실무 처리 기준입니다. 계약
담당공무원은 규정된 바에 따라 신의와 성실의 원칙에 입각하여 처리하여야 합니다.";
                let citation = "(계약예규) 공사계약일반조건 제1조(총칙)";
                if (userText.includes("지체상금") || userText.includes("지체")) {
                    responseContent = "계약상대자가 계약서에 정한 준공기한 내에 공사
를 완성하지 아니한 때에는 매 지체일수마다 지체상금률을 계약금액에 곱하여 산출한 금액을 
현금으로 납부하여야 합니다. 다만, 납부할 금액이 계약금액의 100분의 30을 초과하는 경우에
는 100분의 30으로 합니다.";
                    citation = "(계약예규) 공사계약일반조건 제25조(지체상금)";
                } else if (userText.includes("낙찰자") || userText.includes("종합심사")) {
                    responseContent = "종합심사 점수가 최고점인 자를 낙찰자로 결정합
니다. 종합심사 점수가 최고점인 자가 둘 이상인 경우에는 공사수행능력 및 사회적책임 합산
점수가 높은 자, 균형가격에 근접한 자 등의 순으로 낙찰자를 결정합니다.";
                    citation = "(계약예규) 공사계약 종합심사낙찰제 심사기준 제13조(낙찰
자 결정)";
                }
                botBubble.innerHTML = `
                    <div class="bg-blue-900 text-white rounded-full h-8 w-8 flex 
items-center justify-center font-bold text-sm shrink-0">BOT</div>
                    <div class="bg-slate-100 rounded-2xl px-4 py-3 max-w-2xl 
text-sm leading-relaxed text-slate-700 shadow-sm space-y-2">
                        <p>${responseContent}</p>
                        <div class="mt-2 pt-2 border-t border-slate-200 text-xs 
font-semibold text-blue-800 bg-blue-50 px-3 py-1.5 rounded-lg inline-block">
                            📘 근거 조항: ${citation}
                        </div>
                    </div>
                `;
                chatBox.appendChild(botBubble);
                chatBox.scrollTop = chatBox.scrollHeight;
            }, 600);
        }
        function escapeHtml(text) {
            return text.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, 
"&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#039;");
        }
